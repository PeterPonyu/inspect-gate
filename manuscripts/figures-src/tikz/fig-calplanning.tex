\pgfplotsset{
  % Inside NE — clears curves/bars that occupy the lower/left plot body.
  cal/legend inside ne/.style={
    legend style={
      at={(0.98,0.96)}, anchor=north east,
      font=\fontsize{#1}{9}\selectfont, text=black, draw=none,
      fill=white, fill opacity=0.92, text opacity=1,
      inner sep=1.0pt, row sep=0.4pt, legend cell align=left,
      /tikz/every even column/.append style={column sep=4pt},
      /tikz/every node/.append style={text=black},
    },
  },
  cal/legend inside ne/.default=7.5,
  % Inside NW — panel b curves hug the top-right.
  cal/legend inside nw/.style={
    legend style={
      at={(0.02,0.96)}, anchor=north west,
      font=\fontsize{#1}{9}\selectfont, text=black, draw=none,
      fill=white, fill opacity=0.92, text opacity=1,
      inner sep=1.0pt, row sep=0.4pt, legend cell align=left,
      /tikz/every node/.append style={text=black},
    },
  },
  cal/legend inside nw/.default=7.5,
  % Inside NE, single column — panel c: after x≈2 all series sit on ~0, so a
  % narrow NE key clears the drop at x=1 without a wide 2-col block.
  cal/legend inside ne narrow/.style={
    legend style={
      at={(0.99,0.98)}, anchor=north east,
      font=\fontsize{#1}{8.5}\selectfont, text=black, draw=none,
      fill=white, fill opacity=0.94, text opacity=1,
      inner sep=0.8pt, row sep=0.2pt, legend cell align=left,
      /tikz/every node/.append style={text=black},
    },
  },
  cal/legend inside ne narrow/.default=6.5,
  cal/title near/.style={
    title style={
      at={(0.5,1.0)}, anchor=south, yshift=8pt,
      font=\titlefont, text=black,
    },
  },
}
