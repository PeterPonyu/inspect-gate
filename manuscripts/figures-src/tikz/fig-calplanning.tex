% Letter on the title/legend band, parked in the left gutter (south-east →
% sits left of NW). Clears legend@~1.28 / 2-line panel-a key.
\newcommand{\callabel}[2]{\panellabelshift{#1}{#2}{-16pt}{28pt}}

\pgfplotsset{
  cal/title near/.style={
    title style={
      at={(0.5,1.025)}, anchor=south,
      font=\titlefont, text=black,
    },
  },
  % Shared outside-legend chrome: above the title row (house: crcbaseline).
  cal/legend above/.style={
    legend style={
      at={(0.5,1.26)}, anchor=south,
      legend columns=-1, font=\tickfont, text=black, draw=none, fill=none,
      inner sep=1.0pt, row sep=0.3pt, legend cell align=left,
      /tikz/every even column/.append style={column sep=6pt},
      /tikz/every node/.append style={text=black},
    },
  },
  % Panels c/d: multi-entry keys — two columns above keeps width under panel.
  cal/legend above wrap/.style={
    legend style={
      at={(0.5,1.28)}, anchor=south,
      legend columns=2, font=\tickfont, text=black, draw=none, fill=none,
      inner sep=1.0pt, row sep=0.3pt, legend cell align=left,
      /tikz/every even column/.append style={column sep=7pt},
      /tikz/every node/.append style={text=black},
    },
  },
}
