\pgfplotsset{
  % Compact filled square — apply AFTER ybar= so it is not overridden.
  opcost/sq legend/.style={
    legend image code/.code={%
      \draw[#1] (0.05em,-0.12em) rectangle (0.70em,0.48em);%
    },
  },
  opcost/title near/.style={
    title style={
      at={(0.5,1.0)}, anchor=south, yshift=6pt,
      font=\titlefont, text=black,
    },
  },
  % 1-row legend above title. Title top≈16pt; legend south@42pt → clear gap.
  opcost/legend above 1row/.style={
    legend style={
      at={(0.5,1.0)}, anchor=south, yshift=42pt,
      font=\tickfont, text=black, draw=none, fill=none,
      inner sep=0.6pt, cells={anchor=west},
      /tikz/every even column/.append style={column sep=5pt},
      /tikz/every node/.append style={text=black},
    },
  },
  % 2x2 legend inside NW (c) — log headroom above ~1.8e4 bars.
  opcost/legend inside nw/.style={
    legend style={
      at={(0.02,0.98)}, anchor=north west,
      font=\tickfont, text=black, draw=none, fill=none,
      inner sep=1.0pt, cells={anchor=west},
      /tikz/every even column/.append style={column sep=4pt},
      /tikz/every node/.append style={text=black},
    },
  },
}
