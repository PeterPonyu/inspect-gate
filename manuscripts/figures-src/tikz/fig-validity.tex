% Letter in the left gutter on the title row; wider xshift so horizontal
% legends starting near plot-left cannot cover it.
\newcommand{\validitylabel}[2]{%
  \node[anchor=south east, font=\panellabelfont, inner sep=0.5pt]
    at ([xshift=-18pt, yshift=7pt] #1.north west) {\textbf{#2}};%
}
