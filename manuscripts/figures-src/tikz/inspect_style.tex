% Typography defaults
\newcommand{\figfont}{\fontsize{8}{9.5}\selectfont}
\newcommand{\axislabelfont}{\fontsize{9}{10}\selectfont}
\newcommand{\titlefont}{\fontsize{9}{10}\selectfont}
\newcommand{\tickfont}{\fontsize{7}{8}\selectfont}
% Bold weight policy (house style): ONLY panel letters (a)/(b)/… may use
% \bfseries. Axis titles, tick labels, legends, row/column headers, in-plot
% annotations, and schematic stage titles stay medium/normal weight.
% Panel-label house style (SSOT for all multi-panel Inspect figures):
%   format  (a)  lowercase letter in parentheses
%   font    Computer Modern / figure default, 9 pt, bold, black
%   park    left gutter at axis NW (default xshift=-8pt, yshift=7pt)
% R/Cairo mirrors this via PANEL_LABEL_* in R/_figconst.R.
\newcommand{\panellabelfont}{\fontsize{9}{10}\selectfont\bfseries}
\newcommand{\panellabeltext}[1]{(#1)}

% Panel label OUTSIDE the axis (axis clip would otherwise swallow it).
% Usage after \end{axis}: \panellabel{axname}{a}
% Parked in the left gutter at the top of the axis frame — NOT on the
% title baseline — so titles can stay truly centered over the plot box.
% Custom gutter offsets: \panellabelshift{axname}{a}{-16pt}{11pt}
\newcommand{\panellabelshift}[4]{%
  \node[anchor=south east, font=\panellabelfont, text=black, inner sep=0.5pt]
    at ([xshift=#3, yshift=#4] #1.north west) {\panellabeltext{#2}};%
}
\newcommand{\panellabel}[2]{\panellabelshift{#1}{#2}{-8pt}{7pt}}

% Standard axis styles. Titles centered above the plot box only.
% Text colour policy: axis titles, tick labels, legends, and (via
% \panellabel) panel letters default to black — never grey / muted.
\pgfplotsset{
  inspect/centertitle/.style={
    title style={at={(0.5,1.10)}, anchor=south, font=\titlefont, text=black},
  },
  inspect default/.style={
    axis x line=bottom,
    axis y line=left,
    xlabel style={font=\axislabelfont, text=black},
    ylabel style={font=\axislabelfont, text=black},
    inspect/centertitle,
    x tick label style={font=\tickfont, text=black},
    y tick label style={font=\tickfont, text=black},
    legend style={font=\tickfont, text=black, draw=none, fill=none,
      /tikz/every node/.append style={text=black}},
  }
}
