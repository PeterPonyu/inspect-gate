% CERTIFY gate chip: left accent + tinted fill + lining G# (not math $G_1$).
% Three-line stack — code / descriptor / threshold — so G1/G2 read at the
% same optical size as nearby body text (no tiny subscripts) and "false-rej."
% is not cramped beside the code. Weight from accent / tint / size hierarchy;
% house style keeps medium weight (no \bfseries outside panel letters).
% #1 node name, #2 accent color, #3 code (G1/G2), #4 descriptor, #5 threshold.
\newcommand{\gatechip}[5]{%
  \fill[#2,rounded corners=0.25mm]
    ([xshift=-12.35mm,yshift=10.6mm]#1.center) rectangle
    ([xshift=-10.70mm,yshift=-10.6mm]#1.center);%
  \node[font=\Large,text=#2,anchor=west,inner sep=0pt]
    at ([xshift=-8.7mm,yshift=5.4mm]#1.center) {#3};%
  \node[font=\large,text=FAink,anchor=west,inner sep=0pt]
    at ([xshift=-8.7mm,yshift=0.05mm]#1.center) {#4};%
  \node[font=\normalsize,text=FAinktwo,anchor=west,inner sep=0pt]
    at ([xshift=-8.7mm,yshift=-5.35mm]#1.center) {#5};%
}
