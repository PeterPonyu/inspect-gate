% Draw a normalized heat cell. #1 x, #2 y, #3 value/display pair, #4 semantic color.
\newcommand{\heatcell}[4]{%
  \def\splitpair##1/##2\relax{\def\rawvalue{##1}\def\celldisplay{##2}}%
  \expandafter\splitpair#3/\relax
  \ifx\rawvalue\empty
    \fill[missingcell] (#1,#2) rectangle ++(\cellw,\rowpitch);%
    \node[font=\rmfamily\fontsize{9.0}{9.0}\selectfont, text=missingtext] at ($(#1,#2)+(0.5*\cellw,0.5*\rowpitch)$) {\celldisplay};%
  \else
    \pgfmathtruncatemacro{\shade}{min(100,max(0,round(\rawvalue)))}%
    \colorlet{heatfill}{#4!\shade!white}%
    \fill[heatfill] (#1,#2) rectangle ++(\cellw,\rowpitch);%
    \pgfmathparse{\rawvalue > 55 ? 1 : 0}%
    \ifnum\pgfmathresult=1\def\heattext{white}\else\def\heattext{black}\fi
    \node[font=\rmfamily\fontsize{9.0}{9.0}\selectfont, text=\heattext] at ($(#1,#2)+(0.5*\cellw,0.5*\rowpitch)$) {\celldisplay};%
  \fi
  \draw[white,line width=0.35pt] (#1,#2) rectangle ++(\cellw,\rowpitch);%
}

% Row args: sep, category, type, 4 rates, then 4 counts gobbled so they stay
% out of the typesetting stream (counts live in the TSV/digest only).
% #1=1 draws a category separator at the top of this row.
\newcommand{\gobblefour}[4]{}
\newcommand{\mondrianrow}[7]{%
  \pgfmathsetmacro{\yrow}{\figureheight-\topline-(\rowindex+1)*\rowpitch}%
  \node[anchor=east, font=\rmfamily\fontsize{9.0}{9.0}\selectfont, inner sep=0pt] at (\leftgutter-0.05,\yrow+0.5*\rowpitch) {\labeltext{#2}{#3}};%
  \heatcell{\panelone}{\yrow}{#4}{misscolor}%
  \heatcell{\panelone+\cellw}{\yrow}{#5}{defercolor}%
  \heatcell{\paneltwo}{\yrow}{#6}{misscolor}%
  \heatcell{\paneltwo+\cellw}{\yrow}{#7}{defercolor}%
  % Draw category rules AFTER fills so they stay visible across the data width.
  \ifnum#1=1
    \ifnum\rowindex>0
      \draw[white,line width=1.35pt] (\leftgutter,\yrow+\rowpitch) -- (\dataright,\yrow+\rowpitch);%
      \draw[separator,line width=0.85pt] (\leftgutter,\yrow+\rowpitch) -- (\dataright,\yrow+\rowpitch);%
    \fi
  \fi
  \pgfmathtruncatemacro{\nextrow}{\rowindex+1}%
  \renewcommand{\rowindex}{\nextrow}%
  \gobblefour% trailing n_def/n counts from R
}

% Legend packed in [0, \legendband): rule under grid, clearance, then ticks+swatches.
% 0%/100% must stay strictly below the last data row (y=\legendband).
\newcommand{\drawlegend}{%
  \def\legrowlabx{\leftgutter}%
  \def\legswatchx{\leftgutter+0.62}%
  \def\legswatchw{0.17}%
  \pgfmathsetmacro{\legsepy}{\legendband-0.06}%
  \draw[black!45,line width=0.40pt] (\leftgutter,\legsepy) -- (\dataright,\legsepy);%
  % Keep scale ticks with the swatch strip (not up against the grid).
  \node[anchor=east,font=\rmfamily\fontsize{9.0}{9.0}\selectfont] at (\legswatchx-0.03,0.40) {0\%};%
  \node[anchor=west,font=\rmfamily\fontsize{9.0}{9.0}\selectfont] at (\legswatchx+6*\legswatchw+0.03,0.40) {100\%};%
  \node[anchor=east,font=\rmfamily\fontsize{10.0}{10.0}\selectfont,text=misscolor] at (\legrowlabx,0.255) {miss};%
  \node[anchor=east,font=\rmfamily\fontsize{10.0}{10.0}\selectfont,text=defercolor] at (\legrowlabx,0.125) {defer};%
  \foreach \p [count=\i from 0] in {0,20,40,60,80,100}{%
    \pgfmathtruncatemacro{\shade}{\p}%
    \colorlet{misslegend}{misscolor!\shade!white}%
    \colorlet{deferlegend}{defercolor!\shade!white}%
    \fill[misslegend] (\legswatchx+\i*\legswatchw,0.21) rectangle ++(\legswatchw,0.095);%
    \fill[deferlegend] (\legswatchx+\i*\legswatchw,0.08) rectangle ++(\legswatchw,0.095);%
  }%
  \node[anchor=west,font=\rmfamily\fontsize{10.0}{10.0}\selectfont] at (\legswatchx+6*\legswatchw+0.42,0.19) {rate (\%)};%
}
